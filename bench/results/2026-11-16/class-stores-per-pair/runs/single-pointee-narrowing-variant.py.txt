// inline points at, or -1 where it holds none or they point at several.
fn life_single_pointee(p: Program&, l: Lifetimes&, ty: i64) -> i64 {
    if ty < 0 { return -1; }
    while l.single_pointee.len < p.types.len { l.single_pointee.push(-2); }
    if l.single_pointee[ty] != -2 { return l.single_pointee[ty]; }
    var work: i64[>..] = [ty]; var at = 0; var only: i64 = -1; var several = false;
    while at < work.len && !several {
        let t: const Type& = p.types[work[at]]; at += 1;
        if t.kind == TY_SLICE || t.kind == TY_REF {
            if t.kind == TY_REF && t.length_storage >= 0 || t.element < 0 { several = true; }
            else if only < 0 { only = t.element; }
            else if !life_same(p, only, t.element) { several = true; }
        } else if t.kind == TY_ARRAY && t.element >= 0 {
            var seen = false; for old in work { if old == t.element { seen = true; } }
            if !seen { work.push(t.element); }
        } else if t.kind == TY_STRUCT || t.kind == TY_ENUM || t.kind == TY_TUPLE {
            var field = t.first;
            while field >= 0 {
                let ft = p.nodes[field].type_id; var seen = ft < 0;
                for old in work { if old == ft { seen = true; } }
                if !seen { work.push(ft); }
                field = p.nodes[field].next;
            }
        }
    }
    if several { only = -1; }
    l.single_pointee[ty] = only;
    return only;
}
