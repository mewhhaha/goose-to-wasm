// leads to: a storage class of the same argument further in, or a storage
// class of one pointee made under it.
fn life_class_within(l: Lifetimes&, inner: i64, outer: i64) -> bool {
    let owner = l.roots[outer].class_frame;
    if owner < 0 || l.roots[inner].class_frame != owner || inner == outer { return false; }
    var up = inner;
    while l.roots[up].class_parent >= 0 { up = l.roots[up].class_parent; if up == outer { return true; } }
    if l.roots[outer].class_parent >= 0 { return false; }
    let base = l.frames[owner].class_base; let a = l.roots[up].class_index; let b = l.roots[outer].class_index;
    return a % base == b % base && a / base > b / base;

// What of a stored set was read straight out of the storage of the one
// argument class root it is stored into (own), or the rest of it:
// `b.items = b.storage[..n]` puts back into b's storage what that storage

// reference of another kind (one into the elements of an array the storage

// call lands it in each storage the argument may be as what that storage
// itself holds (life_replay). Mapped pair by pair as any other store, every
// storage the argument might be took what every other held.
fn life_own_reads(l: Lifetimes&, root: i64, set: i64, own: bool) -> i64 {
    let owner = l.roots[root].class_frame;
    if owner < 0 || set == 0 || l.roots[root].class_parent >= 0 || l.roots[root].class_index >= l.frames[owner].class_base { if own { return 0; } return set; }

        if read == own { life_build_add_alt(l, out, x); }
    }
    let part = life_build_end(l, out);
    if !any { if own { return 0; } return set; }
    return part;
}
// What of a set lies in what `held` names: the alternatives at a root of
// `held`, or at a class root within `place`.
fn life_held_part(l: Lifetimes&, set: i64, held: i64, place: i64) -> i64 {
    if set == 0 { return 0; }
    var out = LifeBuild {}; var at = l.sets[set].first; var dropped = false;
    while at >= 0 {
        var x = l.alts[at]; at = x.next; x.next = -1;

        life_build_add_alt(l, out, x);
    }
    let part = life_build_end(l, out);
    if !dropped { return set; }
    return part;
}
